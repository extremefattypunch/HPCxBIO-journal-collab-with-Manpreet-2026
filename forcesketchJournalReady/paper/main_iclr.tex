% ICLR 2027 build. Shares body.tex verbatim with main.tex (the JCTC build), so a
% correction to the science lands in both manuscripts at once. This file
% contributes only the class, the anonymous title block and the bibliography
% style.
%
% Double-blind: the class supplies the anonymous author block, so there is
% deliberately no \authorblock call here -- that is the one line of main.tex that
% names the author. The body carries no URLs and no names, and it reaches its
% acknowledgement only through \fsAcknowledgements, which this file defines as
% empty (below), so nothing else needs suppressing.
%
% Page budget: ICLR allows 9 pages of main text with unlimited references and
% appendices. The shared body already renders to 9 pages, so nothing is cut.
\documentclass{article}
\input{preamble_iclr}
\input{macros}

\input{artifact}
% ICLR is double-blind. The artifact IS public, but its URL contains the author's
% GitHub handle, so citing it here would de-anonymise the submission. If an
% anonymised mirror exists (\fsArtifactAnon non-empty) it is cited; otherwise the
% statement says so plainly rather than omitting the question, which is what a
% reviewer actually needs to know.
\newcommand{\fsArtifactLink}{%
  \ifx\fsArtifactAnon\empty
    a public repository whose URL is withheld here because it identifies the
    authors under double-blind review. It will be cited in the camera-ready, and
    can be made available to reviewers through the program chairs on request%
  \else
    an anonymised mirror, \url{\fsArtifactAnon}%
  \fi}

% The acknowledgement credits Harvard's FASRC cluster by name, which would
% de-anonymise the submission, so it is empty here rather than absent from the
% body: the shared body calls \fsAcknowledgements unconditionally and every
% wrapper must define it. build_iclr.sh independently greps the rendered pdf for
% "Harvard" and fails if this ever regresses.
\newcommand{\fsAcknowledgements}{}
